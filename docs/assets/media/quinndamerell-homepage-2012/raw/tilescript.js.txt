function getTransformProperty(elm) {
// get browser property
    var properties = ['transform','WebkitTransform','MozTransform','msTransform','OTransform'];
    while (p = properties.shift()) {
		//alert(p);
        if (typeof elm.style[p] != 'undefined') {
            return p;
        }
    }
    return false;
}
